\directlua{
function dump(t)
 local n=0 for _ in ipairs(t) do n=n+1 end
 texio.write_nl("@@R "..n)
 for i=1,n do local x=t[i]
  if x.csname then texio.write_nl("@@"..i.." cs "..x.csname.." "..x.cmdname)
  else texio.write_nl("@@"..i.." "..x.command.." "..x.cmdname.." "..x.mode.." "..x.tok) end
 end
end
}
