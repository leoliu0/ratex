\directlua{
function say(...)
  local a = table.pack(...)
  local t = {}
  for i = 1, a.n do t[i] = tostring(a[i]) end
  tex.sprint(-2, table.concat(t, " "))
  tex.sprint("\\par")
end
xform = 240
xtag = 241
local glyph_id = node.id("glyph")
census = {}
local function transform(head)
  for g in node.traverse_id(glyph_id, head) do
    local mode = node.has_attribute(g, xform)
    local key = tostring(mode)
    census[key] = (census[key] or 0) + 1
    if mode == 1 then
      g.char = g.char - 32
    elseif mode == 2 then
      g.char = g.char + 1
    elseif mode == 3 then
      node.unset_attribute(g, xform)
      node.set_attribute(g, xtag, 9)
    end
  end
  return true
end
local function both(head) return transform(head) end
luatexbase.add_to_callback("pre_linebreak_filter", both, "engine.attr.paragraph")
luatexbase.add_to_callback("hpack_filter", function(head, group) return transform(head) end, "engine.attr.hpack")
}
