\directlua{
function say(...)
  local a = table.pack(...)
  local t = {}
  for i = 1, a.n do t[i] = tostring(a[i]) end
  tex.sprint(-2, table.concat(t, " "))
  tex.sprint("\\par")
end
local GLYPH, KERN, DISC = node.id("glyph"), node.id("kern"), node.id("disc")
function probe(label)
  local b = tex.box[0]
  local glyphs, codes, kerns = 0, {}, 0
  for n in node.traverse(b.list) do
    if n.id == GLYPH then
      glyphs = glyphs + 1
      codes[rawlen(codes) + 1] = n.char
    elseif n.id == KERN then
      kerns = kerns + 1
    elseif n.id == DISC then
      local inner = {}
      for g in node.traverse(n.replace) do inner[rawlen(inner) + 1] = g.char end
      codes[rawlen(codes) + 1] = "d" .. table.concat(inner, "+")
    end
  end
  say(label, "glyphs", glyphs, "kerns", kerns, "width", b.width, "codes", table.concat(codes, " "))
end
}
