function say(...)
  local a = table.pack(...)
  local t = {}
  for i = 1, a.n do t[i] = tostring(a[i]) end
  tex.sprint(-2, table.concat(t, " "))
  tex.sprint("\\par")
end
local png = img.scan{filename = "engine_asset.png"}
say("png", png.width, png.height, png.xres, png.yres)
say("png-size", png.xsize, png.ysize, png.transform, png.rotation)
local scaled = img.scan{filename = "engine_asset.png", width = tex.sp("96bp")}
say("png-scaled", scaled.width, scaled.height, scaled.xsize, scaled.ysize)
local p2 = img.scan{filename = "engine_asset.pdf", page = 2, pagebox = "media"}
say("pdf-p2", p2.pages, p2.page, p2.width, p2.height, p2.xsize, p2.ysize)
local p3 = img.scan{filename = "engine_asset.pdf", page = 3, pagebox = "media"}
say("pdf-p3", p3.page, p3.bbox and p3.bbox[3], p3.bbox and p3.bbox[4])
local crop = img.scan{filename = "engine_asset.pdf", page = 1, pagebox = "crop"}
say("pdf-crop", crop.bbox[1], crop.bbox[2], crop.bbox[3], crop.bbox[4])
}
