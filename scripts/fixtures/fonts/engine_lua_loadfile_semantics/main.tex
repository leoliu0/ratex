function say(...)
  local a = table.pack(...)
  local t = {}
  for i = 1, a.n do t[i] = tostring(a[i]) end
  tex.sprint(-2, table.concat(t, " "))
  tex.sprint("\\par")
end
local function slurp(name)
  local f = assert(io.open(name, "rb"))
  local s = f:read("a")
  f:close()
  return s
end

say("shebang-file", loadfile("engine_shebang.lua")())
say("bom-file", dofile("engine_bom.lua"))
say("bom+shebang-file", dofile("engine_bom_shebang.lua"))

local bom, shebang = slurp("engine_bom.lua"), slurp("engine_shebang.lua")
say("raw-bytes", string.byte(bom, 1, 3))
say("load-string-shebang", load(shebang) == nil, load(shebang, "=chunk") == nil)
say("load-string-bom", load(bom) == nil)
say("load-skip-manually", load(shebang:sub(shebang:find(string.char(10), 1, true), -1))() )
local f, e = load("return 1 +", "=my-chunk")
say("syntax-load", f, string.find(e, "my-chunk:1:", 1, true) ~= nil)

local bad, why = loadfile("engine_syntax.lua")
say("syntax-file", bad, string.find(why, "engine_syntax.lua:", 1, true) ~= nil)
local dok, derr = pcall(dofile, "engine_syntax.lua")
say("syntax-dofile", dok, string.find(derr, "engine_syntax.lua:2:", 1, true) ~= nil)
local nofile, nowhy = loadfile("engine_nothere.lua")

say("binary-default", loadfile("engine_dump.luac")(5, 1), load(dumped, "=d", "t"), load(dumped, "=d", "b")(1, 2))
os.remove("engine_dump.luac")
