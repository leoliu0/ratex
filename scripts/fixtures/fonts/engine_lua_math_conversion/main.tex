function say(...)
  local a = table.pack(...)
  local t = {}
  for i = 1, a.n do t[i] = tostring(a[i]) end
  tex.sprint(-2, table.concat(t, " "))
  tex.sprint("\\par")
end
local types = node.types()
function dump(list, depth)
  local out = {}
  for n in node.traverse(list) do
    local name = node.type(n.id)
    if n.id == node.id("glyph") then
      out[rawlen(out) + 1] = "g" .. n.char
    elseif n.id == node.id("hlist") or n.id == node.id("vlist") then
      out[rawlen(out) + 1] = name .. "(" .. dump(n.list) .. ")"
    elseif n.id == node.id("glue") or n.id == node.id("kern") or n.id == node.id("rule") then
      out[rawlen(out) + 1] = string.sub(name, 1, 1) .. n.subtype
    else
      out[rawlen(out) + 1] = name
    end
  end
  return table.concat(out, " ")
end
function mathbox(label, n)
  local b = tex.box[n]
  say(label, "w", b.width, "h", b.height, "d", b.depth, dump(b.list))
end
}
