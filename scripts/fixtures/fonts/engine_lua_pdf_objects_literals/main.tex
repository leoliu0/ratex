\directlua{
function say(...)
  local a = table.pack(...)
  local t = {}
  for i = 1, a.n do t[i] = tostring(a[i]) end
  tex.sprint(-2, table.concat(t, " "))
  tex.sprint("\\par")
end
local half = pdf.reserveobj()
pdf.immediateobj(half, "<< /Type /ExtGState /ca 0.25 /CA 0.25 >>")
local direct = pdf.immediateobj("<< /Type /ExtGState /ca 1 >>")
pdf.setpageresources("/ExtGState << /EngHalf " .. half .. " 0 R /EngFull " .. direct .. " 0 R >>")
say("objects", half > 0, direct > 0, half ~= direct, direct > half)
local res = pdf.getpageresources()
say("page-resources", res ~= nil, string.find(res, "EngHalf", 1, true) ~= nil, string.find(res, "EngFull", 1, true) ~= nil)
}
