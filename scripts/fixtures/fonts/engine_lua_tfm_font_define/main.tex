\directlua{
function say(...)
  local a = table.pack(...)
  local t = {}
  for i = 1, a.n do t[i] = tostring(a[i]) end
  tex.sprint(-2, table.concat(t, " "))
  tex.sprint("\\par")
end
}
\newcommand\fontinfo[1]{\directlua{
local f = font.getfont(font.current())
say("#1", f.format, f.name, f.size, f.designsize, f.characters[65].width, f.characters[string.byte("f")].width, f.parameters.slant, f.parameters.space)
}}

say("define", newid ~= id, f.characters[code].width)
}
