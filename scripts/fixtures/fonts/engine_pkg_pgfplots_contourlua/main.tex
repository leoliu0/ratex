% LuaTeX computes the level sets with the pgfplots Lua backend; engines
% without \directlua draw the same contours from a prepared table.
\newcommand{\levelsets}{%
  \ifpgfplots@LUA@supported
    \addplot3[contour lua={levels={1,4,9},labels=true},domain=-3.5:3.5,y domain=-3.5:3.5,samples=29] {x^2+y^2};
  \else
    \addplot3[contour prepared={labels=true}] table {contours.dat};
  \fi}
\makeatother
